import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import scipy.cluster.hierarchy as hclust
from seaborn import kdeplot, scatterplot

def nan_replace(x):
    #identificare pozitii elemente lipsa - sub forma de ndarray cu valori booleene (unde lipseste valoare se pune True)
    is_nan = np.isnan(x)
    print("is_nan: ", is_nan)
    print("type is_nan: ", type(is_nan))

    #identificare indecsi pt elementele lipsa
    k_nan = np.where(is_nan) # array cu indexul fiecarui element care lipseste, pus de atatea ori de cate ori se gasesc valori lipsa pt tara resp +
    # + array cu indecsii coloanelor unde sunt lipsurile
    # de ex in primul array e 1 1 1 1 1 1 deci pt prima tara lipsesc 6 factori
    # si in al doilea array sunt 0, 3, 5, 6, 8, 9 adica pozitiile pt factorii unde sunt acele lipsuri
    print("k_nan: ", k_nan)
    print("type k_nan: ", type(k_nan))

    print("k_nan[1]: ", k_nan[1]) # vectorul cu indecsii coloanelor unde s lipsurile
    print("type k_nan[1]: ", type(k_nan[1]))
    x[k_nan] = np.nanmean(x[:, k_nan[1]], axis=0)


def plot_ierarhie(h, instante, titlu, prag=None):
    fig = plt.figure(titlu, figsize=(14, 8))
    ax = fig.add_subplot(1, 1, 1)

    ax.set_title(titlu, fontdict={"fontsize": 16, "color": "b"})

    hclust.dendrogram(h, labels=instante, ax=ax, color_threshold=prag)


# h - analiza cluster
# p - nr de pasi
# metoda - metoda de clusterizare, in cazul acesta WARD
def partitie(h, nr_clusteri, p, instante, metoda):
    # distanta maxima dintre 2 pct
    k_dif_max = p - nr_clusteri
    prag = (h[k_dif_max, 2] + h[k_dif_max + 1, 2]) / 2
    print("h[k_dif_max, 2]: ", h[k_dif_max, 2])
    print("h[k_dif_max + 1, 2]: ", h[k_dif_max + 1, 2])
    print("prag:", prag)

    plot_ierarhie(h, instante, "Partitia cu " + str(nr_clusteri) + " clusteri. Metoda: " + metoda, prag)

    n = p + 1

    # c[i] = clusterul din care face parte instanta i
    c = np.arange(n)
    print("c: ", c)
    for i in range(n - nr_clusteri):
        k1 = h[i, 0]
        k2 = h[i, 1]
        c[c == k1] = n + i
        c[c == k2] = n + i

    coduri = pd.Categorical(c).codes
    return np.array(["c" + str(cod + 1) for cod in coduri])

def histograma(x, variabila, partitia):
    fig = plt.figure(figsize=(14, 8))

    fig.suptitle("Histograme pt variabila " + variabila)

    clusteri = list(set(partitia)) # eliminam duplicatele si construim o lista
    size = len(clusteri)

    axs = fig.subplots(1, size, sharey=True)
    for i in range(size):
        ax = axs[i]
        ax.set_xlabel(clusteri[i])
        ax.hist(x[partitia == clusteri[i]], bins=10, rwidth=0.9, range=(min(x), max(x)))

def plot_distributii(x, variabila, partitia):
    fig = plt.figure(figsize=(9, 9))
    ax = fig.add_subplot(1, 1, 1)
    ax.set_title("Distributii pentru variabila "+variabila)
    kdeplot(x=x, hue=partitia, ax=ax)

def plot_partitie(x, partitia, etichete=None):
    fig = plt.figure(figsize=(9, 9))
    ax = fig.add_subplot(1, 1, 1)
    ax.set_title("Plot partitie")
    scatterplot(x=x[:, 0], y=x[:, 1], hue=partitia, hue_order=np.unique(partitia), ax=ax)

    if etichete is not None:
        for i in range(len(etichete)):
            ax.text(x[i, 0], x[i, 1], etichete[i])

def show():
    plt.show()
