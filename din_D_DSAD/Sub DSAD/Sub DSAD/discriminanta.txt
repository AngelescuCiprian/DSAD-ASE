import numpy as np
import pandas as pd
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.naive_bayes import GaussianNB
from sklearn.model_selection import train_test_split
from util import *

def execute():
    tabel_train_test = pd.read_csv("dataIN\\hernia.csv", index_col=0)
    nan_replace(tabel_train_test)

    variabile = list(tabel_train_test)
    predictori = variabile[:-1]
    tinta = variabile[len(predictori)]

    print("predictori: ", predictori)
    print("variabila tinta: ", tinta)

    x_train, x_test, y_train, y_test = train_test_split(tabel_train_test[predictori],
                                                        tabel_train_test[tinta],
                                                        test_size=0.4)

    # Model liniar
    model_lda = LinearDiscriminantAnalysis()
    model_lda.fit(x_train, y_train)

    # Calcul axe discriminante
    clase = model_lda.classes_
    q = len(clase)
    m = q - 1  # numar fct discriminante
    print("m = ", m)

    # Calcul scoruri discriminante model liniar (z)!!!
    z = model_lda.transform(x_test)  # !!!! test, nu train!!!!!!
    et_z = ["z" + str(i + 1) for i in range(m)]
    t_z = pd.DataFrame(z, x_test.index, et_z)
    t_z.to_csv("dataOUT\\z.csv")

    # Plot distributii in axa discriminanta
    for i in range(m):
        plot_distributie(z, y_test, i)

    # Plot instante in axe discriminante
    for i in range(m - 1):
        for j in range(i + 1, m):
            plot_instante(z, y_test, clase, i, j)

    # Predictie in set de testare model liniar
    predictie_lda_test = model_lda.predict(x_test)

    # Evaluare model liniar (matricea de confuzie + indicatori de acuratețe) - calcul metrici
    metrici_lda = calcul_metrici(y_test, predictie_lda_test, clase)
    metrici_lda[0].to_csv("dataOUT\\MatConfuzie_lda.csv")
    metrici_lda[1].to_csv("dataOUT\\Acuratete_lda.csv")
    salvare_erori(y_test, predictie_lda_test, tinta, x_test.index, "LDA")

    # Model Bayesian
    model_b = GaussianNB()
    model_b.fit(x_train, y_train)

    # Predictie in set de testare model bayesian
    predictie_b_test = model_b.predict(x_test)

    # Evaluare model bayesian (matricea de confuzie + indicatori de acuratețe) - calcul metrici
    metrici_b = calcul_metrici(y_test, predictie_b_test, clase)
    metrici_b[0].to_csv("dataOUT\\MatConfuzie_b.csv")
    metrici_b[1].to_csv("dataOUT\\Acuratete_bayes.csv")
    salvare_erori(y_test, predictie_b_test, tinta, x_test.index, "Bayes")

    # Predictii in set de aplicare - pt ambele modele
    x_apply = pd.read_csv("dataIN\\hernia_apply.csv", index_col=0)
    predictie_lda = model_lda.predict(x_apply[predictori])
    predictie_b = model_b.predict(x_apply[predictori])

    tabel_predictii = pd.DataFrame(
        data={
            "Predictie LDA": predictie_lda,
            "Predictie Bayes": predictie_b
        }, index=x_apply.index
    )
    tabel_predictii.to_csv("dataOUT\\Predictii.csv")

    show()


if __name__ == "__main__":
    execute()

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.metrics import confusion_matrix, cohen_kappa_score
from seaborn import kdeplot, scatterplot


def nan_replace(t):
    assert isinstance(t, pd.DataFrame)
    variabile = list(t.columns)

    for each in variabile:
        if any(t[each].isna()):
            if pd.api.types.is_numeric_dtype(t[each]):
                t[each].fillna(t[each].mean(), inplace=True)
            else:
                modul = t[each].mode()[0]
                t[each].fillna(modul, inplace=True)


def plot_distributie(z, y, k=0):
    fig = plt.figure(figsize=(9, 9))
    ax = fig.add_subplot(1, 1, 1)
    ax.set_title("Distributie in axa discriminanta " + str(k + 1))
    kdeplot(x=z[:, k], hue=y, fill=True)


def plot_instante(z, y, clase, k1=0, k2=1):
    fig = plt.figure(figsize=(9, 9))
    ax = fig.add_subplot(1, 1, 1)
    ax.set_title("Plot instante in axe discriminante")
    ax.set_xlabel("z" + str(k1+1))
    ax.set_ylabel("z" + str(k2+1))
    scatterplot(x=z[:, k1], y=z[:, k2], hue=y, hue_order=clase, ax=ax)


def calcul_metrici(y, y_,clase):
    c = confusion_matrix(y, y_)
    tabel_c = pd.DataFrame(c, clase, clase)
    tabel_c["Acuratete"] = np.round(np.diag(c)*100/np.sum(c, axis=1), 3)
    acuratete_medie = tabel_c["Acuratete"].mean()
    acuratete_globala = np.round(sum(np.diag(c))*100/len(y), 3)
    index_ck = cohen_kappa_score(y, y_)
    acuratete = pd.DataFrame(
        data={
            "Acuratete globala": [acuratete_globala],
            "Acuratete medie": [acuratete_medie],
            "Index CK": [index_ck]
        }
    )
    return tabel_c, acuratete


def salvare_erori(y, y_, tinta, instante, model):
    tabel = pd.DataFrame(
        data={
            tinta: y,
            "Predictie": y_
        }, index=instante
    )
    tabel[y != y_].to_csv("err_"+model+".csv")


def show():
    plt.show()
